Page roots include it ahead of every other nonce tag, so this one generates the request's CSP nonce.{% endcomment %} function syncThemeColor() { // NOTE: CSS isn't loaded at first paint, so these mirror --color-background in client/src/index.css. document.querySelector('meta[name="theme-color"]').content = document.documentElement.classList.contains('dark') ? '#0a0a0a' : '#f0f1f3'; } (function() { var theme = localStorage.getItem('theme'); if (theme === 'light') { document.documentElement.classList.add('light'); } else if (theme === 'dark') { document.documentElement.classList.add('dark'); } else if (window.matchMedia('(prefers-color-scheme: dark)').matches) { document.documentElement.classList.add('dark'); } syncThemeColor(); })();
